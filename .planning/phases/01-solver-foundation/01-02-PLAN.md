```python
"""
Thin wrapper around py-opw-kinematics normalizing the API.

Design decisions:
- Module-level singleton Robot instance (initialized once at import)
- forward() returns 4x4 numpy array (not RigidTransform)
- inverse() returns list of tuples, already filtered by M-20iD/20 joint limits
- Do NOT use ee_transform in inverse() (Pitfall #3 from RESEARCH.md)
  Instead compute T_base_wrist = T_base_tcp @ T_tcp_wrist in Python before calling
- filter_by_limits uses epsilon=0.001 rad tolerance (Pitfall #2)
- All parameters from config.py (zero magic numbers)
"""
import numpy as np
from scipy.spatial.transform import RigidTransform
from py_opw_kinematics import KinematicModel, Robot
import config

_KINEMATIC_MODEL = KinematicModel(
    a1=config.OPW_A1, a2=config.OPW_A2, b=config.OPW_B,
    c1=config.OPW_C1, c2=config.OPW_C2, c3=config.OPW_C3, c4=config.OPW_C4,
    offsets=config.OPW_JOINT_OFFSETS,
    flip_axes=config.OPW_FLIP_AXES,
)
_ROBOT = Robot(_KINEMATIC_MODEL, degrees=False)

def forward(joints: tuple) -> np.ndarray:
    """FK: 6 joint angles (radians) -> 4x4 numpy SE3 transform."""
    pose = _ROBOT.forward(joints)
    return np.asarray(pose.as_matrix())

def inverse(T_4x4: np.ndarray) -> list:
    """IK: 4x4 SE3 numpy -> list of up to 8 joint configs (radians).
    Returns only solutions within M-20iD/20 joint limits.
    """
    pose = RigidTransform.from_matrix(T_4x4)
    all_solutions = _ROBOT.inverse(pose)
    if all_solutions is None or len(all_solutions) == 0:
        return []
    return filter_by_limits(all_solutions)

def filter_by_limits(solutions: list) -> list:
    """Filter OPW solutions by M-20iD/20 joint limits.
    Epsilon tolerance avoids floating-point boundary rejections (Pitfall #2).
    """
    _EPS = 0.001  # rad
    lowers = [l - _EPS for l in config.JOINT_LIMITS_LOWER_RAD]
    uppers = [u + _EPS for u in config.JOINT_LIMITS_UPPER_RAD]
    valid = []
    for sol in solutions:
        if all(l <= q <= u for q, l, u in zip(sol, lowers, uppers)):
            valid.append(tuple(sol))
    return valid
```

```python
"""
ENV-03: Verify multiprocessing.Pool with spawn start method
works correctly with the OPW solver.

CRITICAL: Must use 'spawn' start method (never 'fork') to avoid
pybind11/PyO3 GIL state corruption (Pitfall #3 from RESEARCH.md).
The spawn method re-imports modules in each worker process, ensuring
clean C extension state.
"""
import multiprocessing
import numpy as np
import sys
import os

# Ensure optimizer root on path
sys.path.insert(0, os.path.join(os.path.dirname(__file__), ".."))


def _worker_ik(joints_tuple):
    """Worker function that imports solver fresh (spawn re-imports)."""
    from opw_solver.wrapper import forward, inverse
    T = forward(joints_tuple)
    solutions = inverse(T)
    return len(solutions)


def test_spawn_pool():
    """ENV-03: spawn multiprocessing pool with IK solver works without crash.

    Creates a Pool(2) with spawn context, dispatches 20 FK->IK tasks,
    verifies all return non-negative solution counts.
    """
    import config
    import random

    rng = random.Random(42)
    tasks = []
    for _ in range(20):
        joints = tuple(
            rng.uniform(lo, hi)
            for lo, hi in zip(config.JOINT_LIMITS_LOWER_RAD, config.JOINT_LIMITS_UPPER_RAD)
        )
        tasks.append(joints)

    ctx = multiprocessing.get_context("spawn")
    with ctx.Pool(2) as pool:
        results = pool.map(_worker_ik, tasks)

    assert len(results) == 20, f"Expected 20 results, got {len(results)}"
    assert all(r >= 0 for r in results), f"Negative solution count in results: {results}"
    assert any(r > 0 for r in results), f"All 20 tasks returned 0 solutions -- solver likely broken"


def test_spawn_start_method_enforced():
    """Verify that 'spawn' is usable as multiprocessing start method."""
    ctx = multiprocessing.get_context("spawn")
    assert ctx.get_start_method() == "spawn"
```
