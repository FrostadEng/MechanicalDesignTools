```python
"""
EDEN Cell Optimizer -- Dual-Unit Logging Utility

All physical quantities emitted by the optimizer use log_dual() to show
both SI and Imperial values. No ad-hoc print statements for physical quantities.
"""
import logging
import math

_logger = logging.getLogger("optimizer")

def log_dual(label: str, si_value: float, si_unit: str,
             imperial_value: float, imperial_unit: str,
             level: int = logging.INFO) -> None:
    """Log a physical quantity in both SI and Imperial units."""
    msg = f"{label}: {si_value:.3f} {si_unit} ({imperial_value:.3f} {imperial_unit})"
    _logger.log(level, msg)

def mm_to_in(mm: float) -> float:
    return mm / 25.4

def in_to_mm(in_: float) -> float:
    return in_ * 25.4

def kg_to_lb(kg: float) -> float:
    return kg * 2.20462

def lb_to_kg(lb: float) -> float:
    return lb / 2.20462

def rad_to_deg(rad: float) -> float:
    return math.degrees(rad)

def deg_to_rad(deg: float) -> float:
    return math.radians(deg)
```
